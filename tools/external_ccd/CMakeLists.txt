cmake_minimum_required(VERSION 3.18)
project(KyotoExternal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(SCALABLE_CCD_WITH_CUDA OFF CACHE BOOL "" FORCE)
set(SCALABLE_CCD_USE_DOUBLE ON CACHE BOOL "" FORCE)
set(SCALABLE_CCD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SCALABLE_CCD_WITH_PROFILER OFF CACHE BOOL "" FORCE)
add_subdirectory("${UPSTREAM_DIR}" scalable)
add_executable(kyoto_external ../../apps/external_scalable_ccd.cpp)
target_include_directories(kyoto_external PRIVATE ../../include)
target_link_libraries(kyoto_external PRIVATE scalable_ccd::scalable_ccd TBB::tbb)
target_compile_options(kyoto_external PRIVATE -ffp-contract=off -fno-fast-math)
